```sql
        FROM primary_subs
    ) s
    WHERE rn = 1
),
tot AS (
    SELECT count(*) AS n_claims, sum(charge_amount) AS billed
    FROM latest_primary
)
SELECT l.payer_class,
       count(*) AS claims,
       round(100.0 * count(*) / max(tot.n_claims), 1) AS claim_pct,
       round(sum(l.charge_amount), 2) AS billed_charge,
       round(100.0 * sum(l.charge_amount) / nullif(max(tot.billed), 0), 1) AS billed_pct
FROM latest_primary l
CROSS JOIN tot
GROUP BY l.payer_class
ORDER BY l.payer_class
```
